import java.io.*;
import java.util.*;

public class Solution {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        int n = sc.nextInt();
        List<List<Integer>> lines = new ArrayList<>();
        
        // Read the N lines of input
        for (int i = 0; i < n; i++) {
            int d = sc.nextInt();
            List<Integer> line = new ArrayList<>();
            for (int j = 0; j < d; j++) {
                line.add(sc.nextInt());
            }
            lines.add(line);
        }
        
        // Read the number of queries
        int q = sc.nextInt();
        
        // Process each query (x, y)
        for (int i = 0; i < q; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            
            // Convert to 0-based indexing
            int lineIndex = x - 1;
            int elementIndex = y - 1;
            
            // Bounds check
            if (lineIndex >= 0 && lineIndex < lines.size()) {
                List<Integer> currentLine = lines.get(lineIndex);
                if (elementIndex >= 0 && elementIndex < currentLine.size()) {
                    System.out.println(currentLine.get(elementIndex));
                } else {
                    System.out.println("ERROR!");
                }
            } else {
                System.out.println("ERROR!");
            }
        }
        
        sc.close();
    }
}
