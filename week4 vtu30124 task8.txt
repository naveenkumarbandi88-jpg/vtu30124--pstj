class Solution {
    public int[][] transpose(int[][] matrix) {
        int m = matrix.length;        // Number of rows
        int n = matrix[0].length;     // Number of columns

        // Transposed matrix has dimensions n x m
        int[][] result = new int[n][m];

        for (int r = 0; r < m; r++) {
            for (int c = 0; c < n; c++) {
                result[c][r] = matrix[r][c];
            }
        }

        return result;
    }
}

Input
matrix =
[[1,2,3],[4,5,6],[7,8,9]]
Output
[[1,4,7],[2,5,8],[3,6,9]]
Expected
[[1,4,7],[2,5,8],[3,6,9]]
